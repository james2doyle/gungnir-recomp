cmake_minimum_required(VERSION 3.22.1)
project(gungnir_android LANGUAGES C CXX)

if(NOT DEFINED GUNGNIR_ROOT OR GUNGNIR_ROOT STREQUAL "")
    message(FATAL_ERROR "GUNGNIR_ROOT is required")
endif()
if(NOT DEFINED GUNGNIR_SDL3_DIR OR GUNGNIR_SDL3_DIR STREQUAL "")
    message(FATAL_ERROR "GUNGNIR_SDL3_DIR is required")
endif()

set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${GUNGNIR_SDL3_DIR}" "${CMAKE_BINARY_DIR}/sdl3")

include("${GUNGNIR_ROOT}/cmake/AndroidRuntime.cmake")
gungnir_add_runtime_library(main ROOT "${GUNGNIR_ROOT}" SDL3_DIR "${GUNGNIR_SDL3_DIR}")
